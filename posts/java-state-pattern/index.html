import java.awt.BorderLayout;

new Field();

class Light {
private static int ON = 0;
private static int OFF = 1;
private int state;
public Light() {
		state = OFF;
}
public String on_button_pushed() {
if (state == ON) {
			System.out.println("반응 없음");
} else {
			System.out.println("Light On");
			state = ON;
}
return ImageService.lightImageName[0];
}
public String off_button_pushed() {
if (state == OFF)
			System.out.println("반응 없음");
else {
			System.out.println("light off!");
			state = OFF;
}
return ImageService.lightImageName[1];
}

class ImageService {
public static final String[] lightImageName = { "on.png", "off.png", "sleeping.png" };
}
class Field extends JFrame {
private static final long serialVersionUID = 1L;
private Controller controller;
	JLabel lbl;

	Light light;
public Field() {
super("Field");
setDefaultCloseOperation(EXIT_ON_CLOSE);
setLayout(new BorderLayout());

		light = new Light();

		lbl = new JLabel();
		lbl.setIcon(new ImageIcon("./src/img/" + ImageService.lightImageName[1]));

		JPanel jp = new JPanel();
		jp.add(lbl, BorderLayout.CENTER);
add(jp, BorderLayout.CENTER);
		controller = new Controller();
add(controller, BorderLayout.SOUTH);
setSize(400, 300);
setVisible(true);
}
class Controller extends JPanel implements ActionListener {
private static final long serialVersionUID = 1L;
private JButton btnOn, btnOff;
public Controller() {
setLayout(new FlowLayout());
			btnOn = new JButton("On");
			btnOff = new JButton("Off");
			btnOn.addActionListener(this);
			btnOff.addActionListener(this);
add(btnOn);
add(btnOff);
}

		@Override
		public void actionPerformed(ActionEvent e) {
// TODO Auto-generated method stub
			JButton btn = (JButton) e.getSource();
if (btn == btnOn) {
				lbl.setIcon(new ImageIcon("./src/img/" + light.on_button_pushed()));
} else if (btn == btnOff) {
				lbl.setIcon(new ImageIcon("./src/img/" + light.off_button_pushed()));
}
}
}
}

[스테이트 패턴으로 코드 수정 1]



interface State {
public String on_button_pushed(Light light);
public String off_button_pushed(Light light);
}

interface State 에서

State On과 Off 버튼에 대한 동작을 정의 하는 코드를 정의한다.

public String on_button_pushed (Light light);

라이트의 바뀐 상태를 설정해 줘야 하니까

매개 변수로 light 객체를 받는다.



class ON implements State {

	@Override
	public String on_button_pushed(Light light) {
		System.out.println("반응없음");
return ImageService.lightImageName[0];
}

	@Override
	public String off_button_pushed(Light light) {
		light.setState(new OFF());
		System.out.println("불 꺼요");
return ImageService.lightImageName[1];
}
}
class OFF implements State {

	@Override
	public String on_button_pushed(Light light) {
		light.setState(new ON());
		System.out.println("불 켜요");
return ImageService.lightImageName[0];
}

	@Override
	public String off_button_pushed(Light light) {
		System.out.println("반응없음");
return ImageService.lightImageName[1];
}
}

class Light {
private State state;
public Light() {
		state = new OFF();
}
public String on_button_pushed() {
return state.on_button_pushed(this);
}
public String off_button_pushed() {
return state.off_button_pushed(this);
}
public void setState(State state) {
this.state = state;
}
}

State 쪽은 전혀 손을 댈 필요가 없고..

위와 같이 무드등 상태로 진입하는 부분(다음 상태로 변경되는 부분)의 코드만 수정했다.

그런데 지금 상태로 변경 할 때마다 new로 객체를 만들어 내고 있다.

메모리를 할당하고 있는 상황..

이 이부분은 싱글턴 패턴을 사용해 조금 더 수정해 주면 좋다.

싱글턴 패턴 적용



class ON implements State {
private static ON on;
private ON() {
}
public static ON getInstance() {
if (on == null) {
			on = new ON();
}
return on;
}

	@Override
	public String on_button_pushed(Light light) {
		light.setState(SLEEPING.getInstance());
		System.out.println("쿨쿠울");
return ImageService.lightImageName[2];
}

	@Override
	public String off_button_pushed(Light light) {
		light.setState(OFF.getInstance());
		System.out.println("불 꺼요");
return ImageService.lightImageName[1];
}
}

스테이트 패턴, 스트레티지 패턴 차이점

스트레티지 패턴은 전략을 수정하기 위해 사용된다.

예를들어 게임 캐릭터 각각의 설정들,

다른 알고리즘으로 손쉽게 교체하는 동작에 자주 쓰인다.

이와 조금 다르게 스테이트 패턴은,

상태에 따라 다양한 행동을 처리할 때 사용한다.

행위 자체도 상태 객체에 위임을 해서 처리하고 있다.
